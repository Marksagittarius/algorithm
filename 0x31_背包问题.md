# 0x31 背包问题

## 引言

背包问题是最为基础的动态规划问题，常见的背包模型有 01 背包和完全背包模型。

### 01 背包

```c++
int n, m;
const int LEN = 1e3 + 5;
int t[LEN], v[LEN], f[LEN];

int main() {
    scanf("%d %d", &n, &m);

    for (int i = 0; i < m; i++)
        scanf("%d %d", &t[i], &v[i]);

    for (int i = 0; i < m; i++) {
        for (int j = n; j >= t[i]; j--) {
            f[j] = max(f[j], f[j - t[i]] + v[i]);
        }
    }

    printf("%d", f[n]);
    return 0;
}
```

### 完全背包

```c++
int n, m;
const int LEN = 1e7 + 5;
long long t[LEN], v[LEN], f[LEN];

int main() {
    scanf("%d %d", &n, &m);

    for (int i = 0; i < m; i++)
        scanf("%lld %lld", &t[i], &v[i]);

    for (int i = 0; i < m; i++) {
        for (int j = t[i]; j <= n; j++) {
            f[j] = max(f[j], f[j - t[i]] + v[i]);
        }
    }

    printf("%lld", f[n]);
    return 0;
}
```

## 例题

[322. 零钱兑换](https://leetcode.cn/problems/coin-change/description/)

```c++
class Solution {
public:
    int coinChange(vector<int>& coins, int amount) {
        vector<int> f(amount + 1, amount + 1);
        f[0] = 0;
        for (auto coin : coins) {
            for (int i = coin; i <= amount; i++) {
                f[i] = min(f[i], f[i - coin] + 1);
            }
        }
        return f[amount] == amount + 1 ? -1 : f[amount];
    }
};
```

[416. 分割等和子集](https://leetcode.cn/problems/partition-equal-subset-sum/description/)

```c++
class Solution {
public:
    bool canPartition(vector<int>& nums) {
        int sum = 0;
        for (auto& num : nums) sum += num;
        if (sum % 2 == 1) return false;
        vector<bool> f(sum / 2 + 1, false);
        f[0] = true;
        for (auto& num : nums) {
            for (int i = sum / 2; i >= num; i--) {
                f[i] = f[i] || f[i - num];
            }
        }
        return f[sum / 2];
    }
};
```

[494. 目标和](https://leetcode.cn/problems/target-sum/description/)

```c++
class Solution {
public:
    int findTargetSumWays(vector<int>& nums, int target) {
        int sum = 0;
        for (auto& num : nums) sum += num;
        int tar = target + sum;
        if (tar & 1) return false;
        vector<int> f(tar + 1, 0);
        f[0] = 1;
        for (auto& num : nums) {
            for (int i = tar; i >= num; i--) {
                f[i] = f[i] + f[i - num];
            }
        }
        return f[tar / 2];
    }
};
```

```c++
class Solution {
public:
    int change(int amount, vector<int>& coins) {
        vector<int> f(amount + 1, 0);
        f[0] = 1;
        for (auto coin : coins) {
            for (int j = coin; j <= amount; j++) {
                f[j] += f[j - coin];
            }
        }

        return f[amount];
    }
};
```

[474. 一和零](https://leetcode.cn/problems/ones-and-zeroes/description/?envType=study-plan-v2&envId=dynamic-programming)

```c++
class Solution {
public:
    int findMaxForm(vector<string>& strs, int m, int n) {
        int len = strs.size();
        vector<pair<int, int>> cnt;
        for (int i = 0; i < len; i++)
            cnt.push_back(get(strs[i]));
        
        vector<vector<int>> f(m + 1, vector<int>(n + 1, 0));
        for (int i = 0; i < len; i++) {
            for (int j = m; j >= cnt[i].first; j--) {
                for (int k = n; k >= cnt[i].second; k--) {
                    f[j][k] = max(f[j - cnt[i].first][k - cnt[i].second] + 1, f[j][k]);
                }
            }
        }

        return f[m][n];
    }

    pair<int, int> get(string& str) {
        int lhs = 0;
        int rhs = 0;
        
        for (auto& ch : str)
            if (ch == '0')
                lhs++;
            else
                rhs++;
            
        return { lhs, rhs };
    }
};
```

```c++
class Solution {
public:
    int combinationSum4(vector<int>& nums, int target) {
        int n = nums.size();
        vector<int> f(target + 1, 0);
        f[0] = 1;
        for (int i = 1; i <= target; i++) {
            for (auto& num : nums) {
                if (num <= i) {
                    f[i] += f[i - num];
                }
            }
        }
        return f[target];
    }
};
```
