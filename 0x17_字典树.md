# 0x17 字典树

## 引言

字典树是一棵挂满了字符串的树。

字典树也被成为 Trie，其一般构造方法如下：

[208. 实现 Trie (前缀树)](https://leetcode.cn/problems/implement-trie-prefix-tree/)

```c++
class Trie {
public:
    vector<Trie*> children;
    bool end;
    Trie() {
        children = vector<Trie*>(26, nullptr);
        end = false;
    }
    
    void insert(string word) {
        Trie* trie = this;
        for (auto& ch : word) {
            if (!trie->children[ch - 'a'])
                trie->children[ch - 'a'] = new Trie();
            trie = trie->children[ch - 'a'];
        }
        trie->end = true;
    }

    bool search(string word) {
        Trie* trie = this;
        for (auto& ch : word) {
            if (!trie->children[ch - 'a'])
                return false;
            trie = trie->children[ch - 'a'];
        }
        return trie->end;
    }
    
    bool startsWith(string word) {
        Trie* trie = this;
        for (auto& ch : word) {
            if (!trie->children[ch - 'a'])
                return false;
            trie = trie->children[ch - 'a'];
        }
        return true;
    }
};
```

[212. 单词搜索 II](https://leetcode.cn/problems/word-search-ii/description/)

```c++
class Solution {
public:
    class Trie {
    public:
        vector<Trie*> children;
        bool isEnd;
        Trie() : children(26), isEnd(false) {}
        
        void insert(string word) {
            Trie* node = this;
            for (char ch : word) {
                ch -= 'a';
                if (node->children[ch] == nullptr)
                    node->children[ch] = new Trie();
                
                node = node->children[ch];
            }

            node->isEnd = true;
        }
        bool hasChild(char ch) {
            ch -= 'a';
            return this->children[ch] != nullptr;
        }
    };

    Trie* trie;
    set<string> st;
    set<string> aw;
    vector<vector<int> > vis;
    vector<string> findWords(vector<vector<char>>& board, vector<string>& words) {
        int m = board.size();
        int n = board[0].size();
        vis = vector<vector<int> >(m, vector<int>(n));
        trie = new Trie();
        for (auto wd : words) trie->insert(wd);
        for (auto wd : words) aw.insert(wd);
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                dfs(board, i, j, m, n, trie, "");
            }
        }
        vector<string> ans;
        for (auto wd : st) ans.push_back(wd);
        return ans;
    }

    int dx[4] = {0, 1, 0, -1};
    int dy[4] = {1, 0, -1, 0};

    void dfs(vector<vector<char> >& board, int i, int j, int m, int n, Trie* root, string word) {
        if (i < 0 || j < 0 || i >= m || j >= n) return;
        if (vis[i][j]) return;
        char ch = board[i][j];
        if (!root->hasChild(ch)) return;
        word += ch;
        if (aw.count(word)) {
            st.insert(word); 
            aw.erase(word);
        }
        vis[i][j] = 1;
        for (int k = 0; k < 4; k++) {
            int xx = i + dx[k];
            int yy = j + dy[k];
            dfs(board, xx, yy, m, n, root->children[ch - 'a'], word);
        }
        vis[i][j] = 0;
    }
};
```

[648. 单词替换](https://leetcode.cn/problems/replace-words/)

```c++
class Solution {
class Trie {
public:
    vector<Trie*> children;
    bool end;
    Trie() {
        children = vector<Trie*>(26, nullptr);
        end = false;
    }
    
    void insert(string word) {
        Trie* trie = this;
        for (auto& ch : word) {
            if (!trie->children[ch - 'a'])
                trie->children[ch - 'a'] = new Trie();
            trie = trie->children[ch - 'a'];
        }
        trie->end = true;
    }

    string get(string word) {
        Trie* trie = this;
        string str = "";
        for (auto& ch : word) {
            if (trie->end) return str;
            if (!trie->children[ch - 'a'])
                return word;
            str += ch;
            trie = trie->children[ch - 'a'];
        }
        return word;
    }
};

public:
    string replaceWords(vector<string>& dict, string s) {
        Trie* trie = new Trie();
        string ans = "";
        vector<string> arr;
        string str = "";
        s += ' ';
        for (auto& ch : s) {
            if (ch == ' ') {
                arr.push_back(str);
                str = "";
            } else {
                str += ch;
            }
        }
        for (auto& root : dict) {
            trie->insert(root);
        }
        for (int i = 0; i < arr.size(); i++) {
            ans += trie->get(arr[i]);
            if (i != arr.size() - 1)
                ans += ' ';
        }
        return ans;
    }
};
```

[820. 单词的压缩编码](https://leetcode.cn/problems/short-encoding-of-words/)

```c++
class Solution {
public:
    class Trie {
    public:
        vector<Trie*> children;
        bool end;
        Trie() {
            children = vector<Trie*>(26, nullptr);
            end = false;
        }
        
        void insert(string word) {
            Trie* trie = this;
            for (int i = word.size() - 1; i >= 0; i--) {
                char ch = word[i];
                if (!trie->children[ch - 'a'])
                    trie->children[ch - 'a'] = new Trie();
                trie = trie->children[ch - 'a'];
            }
            trie->end = true;
        }

        bool startsWith(string word) {
            Trie* trie = this;
            for (int i = word.size() - 1; i >= 0; i--) {
                char ch = word[i];
                if (!trie->children[ch - 'a'])
                    return false;
                trie = trie->children[ch - 'a'];
            }
            return true;
        }
    };

    int minimumLengthEncoding(vector<string>& words) {
        Trie* trie = new Trie();
        int n = words.size(); 
        sort(words.begin(), words.end(), [&](auto l, auto r) {
            return l.size() > r.size();
        });
        vector<int> mask(n);
        for (int i = 0; i < n; i++) {
            auto word = words[i];
            if (trie->startsWith(word)) {
                mask[i] = 1;
                continue;
            }
            trie->insert(word);
        }
        int ans = 0;
        int cnt = 0;
        for (int i = 0; i < n; i++) {
            if (!mask[i]) {
                ans += words[i].size();
                cnt++;
            }
        }
        return ans + cnt;
    }
};
```
