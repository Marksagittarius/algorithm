# 0x12 栈

## 引言

栈是一种后进先出（LIFO）的数据结构，在计算机科学中具有极其广泛的应用场景。

## 括号问题

[32. 最长有效括号](https://leetcode.cn/problems/longest-valid-parentheses/description/)

```c++
class Solution {
public:
    int longestValidParentheses(string s) {
        int n = s.size();
        int ans = 0;
        stack<int> stk;
        stk.push(-1);
        for (int i = 0; i < n; i++) {
            if (s[i] == '(') {
                stk.push(i);
            } else {
                stk.pop();
                if (!stk.empty()) {
                    ans = max(ans, i - stk.top());
                } else {
                    stk.push(i);
                }
            }
        }
        return ans;
    }
};
```

[394. 字符串解码](https://leetcode.cn/problems/decode-string/description/)

```c++
class Solution {
public:
    bool digit(char& ch) {
        return ch >= '0' && ch <= '9';
    }
    bool letter(char& ch) {
        return ch >= 'a' && ch <= 'z';
    }
    string decodeString(string s) {
        string ans = "";
        stack<int> ntk;
        stack<string> stk;
        int num = 0;
        for (auto& ch : s) {
            if (digit(ch)) {
                num = num * 10 + (ch - '0');
            } else if (letter(ch)) {
                ans += ch;
            } else if (ch == '[') {
                ntk.push(num);
                num = 0;
                stk.push(ans);
                ans = "";
            } else if (ch == ']') {
                int t = ntk.top();
                ntk.pop();
                while (t--) stk.top() += ans;
                ans = stk.top();
                stk.pop();
            }
        }
        return ans;
    }
};
```

## 单调栈

单调栈中的元素能够位置一致的单调性。

[739. 每日温度](https://leetcode.cn/problems/daily-temperatures/description/)

```c++
class Solution {
public:
    vector<int> dailyTemperatures(vector<int>& temperatures) {
        int n = temperatures.size();
        vector<int> ans(n, 0);
        stack<int> stk;
        for (int i = 0; i < n; i++) {
            while (!stk.empty() && temperatures[i] > temperatures[stk.top()]) {
                auto it = stk.top();
                ans[it] = i - it;
                stk.pop();
            }
            stk.push(i);
        }
        return ans;
    }
};
```

基于单调栈能够在线性时间复杂度内计算数组中每个元素[左侧/右侧]第一个[大于/小于]该元素的[下标/值/存在性]。

[84. 柱状图中最大的矩形](https://leetcode.cn/problems/largest-rectangle-in-histogram/description/)

```c++
class Solution {
public:
    int largestRectangleArea(vector<int>& nums) {
        int ans = 0;
        int n = nums.size();
        vector<int> lhs(n, -1);
        vector<int> rhs(n, n);
        stack<int> stk;
        for (int i = 0; i < n; i++) {
            while (!stk.empty() && nums[i] <= nums[stk.top()]) {
                rhs[stk.top()] = i;
                stk.pop();
            }
            lhs[i] = stk.empty() ? -1 : stk.top();
            stk.push(i);
        }
        for (int i = 0; i < n; i++) {
            ans = max(ans, nums[i] * (rhs[i] - lhs[i] - 1));
        }
        return ans;
    }
};
```

[85. 最大矩形](https://leetcode.cn/problems/maximal-rectangle/description/)

```c++
class Solution {
public:
    int maximalRectangle(vector<string>& matrix) {
        int ans = 0;
        int m = matrix.size();
        if (m == 0) return 0;
        int n = matrix[0].size();
        if (n == 0) return 0;
        vector<vector<int>> nums(m, vector<int>(n));
        for (int i = 0; i < n; i++)
            nums[0][i] = matrix[0][i] - '0';
        for (int i = 1; i < m; i++) {
            for (int j = 0; j < n; j++) {
                int num = matrix[i][j] - '0';
                if (num == 1) {
                    nums[i][j] = nums[i - 1][j] + 1;
                }
            }
        }
        for (int i = 0; i < m; i++) {
            vector<int> lhs(n, -1);
            vector<int> rhs(n, n);
            stack<int> stk;
            for (int j = 0; j < n; j++) {
                while (!stk.empty() && nums[i][j] <= nums[i][stk.top()]) {
                    rhs[stk.top()] = j;
                    stk.pop();
                }
                lhs[j] = stk.empty() ? -1 : stk.top();
                stk.push(j);
            }        
            for (int j = 0; j < n; j++) {
                ans = max(ans, nums[i][j] * (rhs[j] - lhs[j] - 1));
            }
        }

        return ans;
    }
};
```

借助单调栈也能够解决构造最小字典序定长子序列问题：

[402. 移掉 K 位数字](https://leetcode.cn/problems/remove-k-digits/description/)

```c++
class Solution {
public:
    string removeKdigits(string num, int k) {
        int n = num.size();
        if (n <= k) return "0";
        string stk = "";   
        for (int i = 0; i < n; i++) {
            while (!stk.empty() && stk.back() > num[i] && k > 0) {
                stk.pop_back();
                k--;
            }
            stk.push_back(num[i]);
        }
        while (k) {
            stk.pop_back();
            k--;
        }
        string ans = "";
        bool flag = true;
        for (auto& ch : stk) {
            if (flag && ch == '0') continue;
            flag = false;
            ans += ch;
        }
        return ans == "" ? "0" : ans;
    }
};
```

[316. 去除重复字母](https://leetcode.cn/problems/remove-duplicate-letters/description/)

```c++
class Solution {
public:
    string removeDuplicateLetters(string s) {
        int n = s.size();
        vector<int> vis(26);
        vector<int> cnt(26);
        string stk;
        for (auto ch : s) cnt[ch - 'a']++;
        for (auto ch : s) {
            if (!vis[ch - 'a']) {
                while (!stk.empty() && stk.back() > ch) {
                    if (cnt[stk.back() - 'a'] > 0) {
                        vis[stk.back() - 'a'] = 0;
                        stk.pop_back();
                    } else {
                        break;
                    }
                }
                vis[ch - 'a'] = 1;
                stk.push_back(ch);
            }
            cnt[ch - 'a']--;
        }
        return stk;
    }
};
```

## 计算器

在算法面试中构造一个功能有限的计算器是面试者常被考查的问题。

[224. 基本计算器](https://leetcode.cn/problems/basic-calculator/description/)

```c++
class Solution {
public:
    int calculate(string s) {
        int n = s.size();
        stack<int> ops;
        int i = 0;
        int ans = 0;
        int sign = 1;
        ops.push(1);
        while (i < n) {
            if (s[i] == ' ') {
                i++;
            } else if (s[i] == '+') {
                sign = ops.top();
                i++;
            } else if (s[i] == '-') {
                sign = -1 * ops.top();
                i++;
            } else if (s[i] == '(') {
                ops.push(sign);
                i++;
            } else if (s[i] == ')') {
                ops.pop();
                i++;
            } else {
                long num = 0;
                while (i < n && s[i] >= '0' && s[i] <= '9') {
                    num = num * 10 + s[i] - '0';
                    i++;
                }
                ans += num * sign;
            }
        }
        return ans;
    }
};
```

[227. 基本计算器 II](https://leetcode.cn/problems/basic-calculator-ii/description/)

```c++
class Solution {
public:
    bool digit(char ch) {
        return ch >= '0' && ch <= '9';
    }

    int calculate(string s) {
        int n = s.size();
        stack<long> stk;
        int ans = 0;
        long num = 0;
        char op = '+';
        for (int i = 0; i < n; i++) {
            if (digit(s[i])) {
                num = num * 10 + s[i] - '0';
            }
            if (!digit(s[i]) && s[i] != ' ' || i == n - 1) {
                if (op == '+') {
                    stk.push(num);
                } else if (op == '-') {
                    stk.push(-num);
                } else if (op == '*') {
                    int t = stk.top();
                    stk.pop();
                    stk.push(t * num);
                } else if (op == '/') {
                    int t = stk.top();
                    stk.pop();
                    stk.push(t / num);
                }
                num = 0;
                op = s[i];
            }
        }
        while (!stk.empty()) {
            ans += stk.top();
            stk.pop();
        }
        return ans;
    }
};
```

如果将上述两题的思路结合可以构造一个支持四则运算和括号运算的计算器：

```c++
class Solution {
public:
    int calculate(string s) {
        int i = 0;
        s += '+';
        return calculate(s, i);
    }

    bool digit(char& ch) {
        return ch >= '0' && ch <= '9';
    }

    int calculate(string& s, int& i) {
        int n = s.size();
        vector<long> stk;
        stk.push_back(0);
        char op = '+';
        long num = 0;
        for (; i < n; i++) {
            char ch = s[i];
            if (ch == ' ') continue;
            if (digit(ch)) num = num * 10 + (ch - '0');
            if (ch == '(') num = calculate(s, ++i);
            if (!digit(ch) || i == n - 1) {
                if (op == '+') stk.push_back(num);
                if (op == '-') stk.push_back(-num);
                if (op == '*') stk.back() *= num;
                if (op == '/') stk.back() /= num;
                op = ch;
                num = 0;
            }
            if (ch == ')') break;
        }
        return accumulate(stk.begin(), stk.end(), 0);
    }
};
```
