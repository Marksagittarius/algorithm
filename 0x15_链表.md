# 链表

## 引言

链表是一种链式存储的数据结构。

软件开发者如果想要创造出 Homebrew 般改变世界的软件，需要掌握[206. 反转链表](https://leetcode.cn/problems/reverse-linked-list/description/)

```c++
class Solution {
public:
    ListNode* reverseList(ListNode* head) {
        if (!head) return head;
        if (!head->next) return head;
        auto ans = reverseList(head->next);
        head->next->next = head;
        head->next = nullptr;
        return ans;
    }
};
```

[234. 回文链表](https://leetcode.cn/problems/palindrome-linked-list/description/)

```c++
class Solution {
public:
    ListNode* front;
    bool f(ListNode* node) {
        if (node) {
            if (!f(node->next)) return false;
            if (node->val != front->val) return false;
            front = front->next;
        }
        return true;
    }
    bool isPalindrome(ListNode* head) {
        front = head;
        return f(head);
    }
};
```

[160. 相交链表](https://leetcode.cn/problems/intersection-of-two-linked-lists/description/)

```c++
class Solution {
public:
    ListNode *getIntersectionNode(ListNode *headA, ListNode *headB) {
        auto l = headA;
        auto r = headB;
        while (l || r) {
            if (l == r) return l;
            l = l ? l->next : headB;
            r = r ? r->next : headA;
        }
        return NULL;
    }
};
```

[141. 环形链表](https://leetcode.cn/problems/linked-list-cycle/description/)

```c++
class Solution {
public:
    bool hasCycle(ListNode *head) {
        if (!head) return false;
        ListNode* fast = head;
        ListNode* slow = head;
        while (slow && fast) {
            slow = slow->next;
            if (!fast->next) return false;
            fast = fast->next->next;
            if (fast == slow) return true;
        }
        return false;
    }
};
```

[142. 环形链表 II](https://leetcode.cn/problems/linked-list-cycle-ii/description/)

```c++
class Solution {
public:
    ListNode *detectCycle(ListNode *head) {
        if (!head) return NULL;
        ListNode* fast = head;
        ListNode* slow = head;
        while (slow && fast) {
            slow = slow->next;
            if (fast->next == NULL) return NULL;
            fast = fast->next->next;
            if (fast == slow) {
                ListNode* ans = head;
                while (slow != ans) {
                    slow = slow->next;
                    ans = ans->next;
                }
                return ans;
            }
        }
        return NULL;
    }
};
```

[19. 删除链表的倒数第 N 个结点](https://leetcode.cn/problems/remove-nth-node-from-end-of-list/description/)

```c++
class Solution {
public:
    ListNode* removeNthFromEnd(ListNode* head, int n) {
        auto h = new ListNode();
        h->next = head;
        auto fast = h; 
        auto slow = h;
        auto prev = h;
        for (int i = 0; i < n; i++)
            fast = fast->next;
        while (fast) {
            fast = fast->next;
            slow = slow->next;
        }
        while (prev->next != slow) {
            prev = prev->next;
        }
        prev->next = slow->next;
        return h->next;
    }
};
```

[25. K 个一组翻转链表](https://leetcode.cn/problems/reverse-nodes-in-k-group/)

```c++
class Solution {
public:
    ListNode* reverse(ListNode* slow, ListNode* fast) {
        if (fast == slow) return fast;
        reverse(slow->next, fast);
        slow->next->next = slow;
        slow->next = nullptr;
        return fast;
    }

    ListNode* reverseKGroup(ListNode* head, int k) {
        if (!head) return head;
        ListNode* h = new ListNode();
        h->next = head;
        ListNode* slow = h;
        ListNode* fast = h;
        while (fast) {
            int l = k;
            while (fast && l) {
                fast = fast->next;
                l--;
            }
            if (!fast) break;
            auto end = fast->next;
            auto start = slow->next;
            reverse(start, fast);
            slow->next = fast;
            start->next = end;
            slow = start;
            fast = start;
        }
        return h->next;
    }
};
```

[148. 排序链表](https://leetcode.cn/problems/sort-list/description/)

```c++
class Solution {
public:
    ListNode* sortList(ListNode* head) {
        if (!head) return head;
        if (!head->next) return head;
        return sort(head, nullptr);
    }

    ListNode* sort(ListNode* lhs, ListNode* rhs) {
        if (!lhs) return lhs;
        if (lhs->next == rhs) {
            lhs->next = nullptr;
            return lhs;
        }
        ListNode* fast = lhs;
        ListNode* slow = lhs;
        while (fast != rhs) {
            slow = slow->next;
            fast = fast->next;
            if (fast != rhs) fast = fast->next;
        }
        auto l = sort(lhs, slow);
        auto r = sort(slow, rhs);
        return merge(l, r);
    }

    ListNode* merge(ListNode* l, ListNode* r) {
        ListNode* h = new ListNode();
        ListNode* p = h;
        ListNode* i = l;
        ListNode* j = r;
        while (i && j) {
            if (i->val < j->val) {
                p->next = i;
                i = i->next;
            } else {
                p->next = j;
                j = j->next;
            }
            p = p->next;
        }
        p->next = i ? i : j;
        return h->next;
    }
};
```

[146. LRU 缓存](https://leetcode.cn/problems/lru-cache/description/)

```c++
class LRUCache {
public:
    class Node {
    public:
        int k, v;
        Node* prev;
        Node* next;
        Node(int k, int v) {
            this->k = k;
            this->v = v;
            this->prev = nullptr;
            this->next = nullptr;
        }
    };
    
    Node* head;
    map<int, Node*> mp;
    int cap; 
    int cnt;

    LRUCache(int capacity) {
        this->cap = capacity;
        this->cnt = 0;
        this->head = new Node(-1, -1);
        head->prev = head;
        head->next = head;
    }

    void insert(Node* head, Node* node) {
        node->next = head->next;
        head->next->prev = node;
        head->next = node;
        node->prev = head;
    }

    void remove(Node* node) {
        node->prev->next = node->next;
        node->next->prev = node->prev;
    }

    int get(int key) {
        if (!mp.count(key)) return -1;
        Node* node = mp[key];
        remove(node);
        insert(head, node);
        return node->v;
    }
    
    void put(int key, int value) {
        if (mp.count(key)) {
            Node* node = mp[key];
            node->v = value;
            remove(node);
            insert(head, node);
        } else {
            Node* node = new Node(key, value);
            if (cap == cnt) {
                Node* rear = head->prev;
                remove(rear);
                mp.erase(rear->k);
                cnt--;
            }
            insert(head, node);
            cnt++;
            mp[key] = node;
        }
    }
};
```

如果需要被构造的并非 LRU 而是 LFU，其思路将迥然不同。
