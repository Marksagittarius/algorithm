## 引言

链表是一种链式存储的数据结构。

软件开发者如果想要创造出 Homebrew 般改变世界的软件，需要掌握[206. 反转链表](https://leetcode.cn/problems/reverse-linked-list/description/)

```c++
class Solution {
public:
    ListNode* reverseList(ListNode* head) {
        if (!head) return head;
        if (!head->next) return head;
        auto ans = reverseList(head->next);
        head->next->next = head;
        head->next = nullptr;
        return ans;
    }
};
```

[234. 回文链表](https://leetcode.cn/problems/palindrome-linked-list/description/)

```c++
class Solution {
public:
    ListNode* front;
    bool f(ListNode* node) {
        if (node) {
            if (!f(node->next)) return false;
            if (node->val != front->val) return false;
            front = front->next;
        }
        return true;
    }
    bool isPalindrome(ListNode* head) {
        front = head;
        return f(head);
    }
};
```

[160. 相交链表](https://leetcode.cn/problems/intersection-of-two-linked-lists/description/)

```c++
class Solution {
public:
    ListNode *getIntersectionNode(ListNode *headA, ListNode *headB) {
        auto l = headA;
        auto r = headB;
        while (l || r) {
            if (l == r) return l;
            l = l ? l->next : headB;
            r = r ? r->next : headA;
        }
        return NULL;
    }
};
```

[141. 环形链表](https://leetcode.cn/problems/linked-list-cycle/description/)

```c++
class Solution {
public:
    bool hasCycle(ListNode *head) {
        if (!head) return false;
        ListNode* fast = head;
        ListNode* slow = head;
        while (slow && fast) {
            slow = slow->next;
            if (!fast->next) return false;
            fast = fast->next->next;
            if (fast == slow) return true;
        }
        return false;
    }
};
```

class Solution {
public:
    ListNode *detectCycle(ListNode *head) {
        if (!head) return NULL;
        ListNode* fast = head;
        ListNode* slow = head;
        while (slow && fast) {
            slow = slow->next;
            if (!fast->next) return NULL;
            fast = fast->next->next;
            if (fast == slow) {
                ListNode* ans = head;
                while (slow != ans) {
                    slow = slow->next;
                    ans = ans->next;
                }
                return ans;
            }
        }
        return NULL;
    }
};

[19. 删除链表的倒数第 N 个结点](https://leetcode.cn/problems/remove-nth-node-from-end-of-list/description/)

```c++
class Solution {
public:
    ListNode* removeNthFromEnd(ListNode* head, int n) {
        ListNode* h = new ListNode();
        h->next = head;
        ListNode* slow = h;
        ListNode* fast = head;
        for (int i = 0; i < n; i++) fast = fast->next;
        while (fast) {
            fast = fast->next;
            slow = slow->next;
        }
        ListNode* node = slow->next;
        slow->next = node->next;
        return h->next;
    }
};
```

[25. K 个一组翻转链表](https://leetcode.cn/problems/reverse-nodes-in-k-group/)

```c++
class Solution {
public:
    ListNode* f(ListNode* l, ListNode* r) {
        if (l == r) return l;
        auto end = f(l->next, r);
        l->next->next = l;
        l->next = nullptr;
        return end;
    }
     
    ListNode* reverseKGroup(ListNode* head, int k) {
        ListNode* h = new ListNode();
        h->next = head;
        ListNode* fast = head; 
        ListNode* slow = h;
        while (fast) {
            int cnt = 0;
            while (fast) {
                cnt++;
                if (cnt == k) break;
                fast = fast->next;
            }
            if (cnt == k) {
                auto start = slow->next;
                auto next = fast->next;
                auto end = f(start, fast);
                slow->next = end;
                start->next = next;
                slow = start;
                fast = next;
            } else {
                break;
            }
        }
        return h->next;
    }
};
```

[82. 删除排序链表中的重复元素 II](https://leetcode.cn/problems/remove-duplicates-from-sorted-list-ii/description/)

```c++
class Solution {
public:
    ListNode* deleteDuplicates(ListNode* head) {
        if (!head) return head;
        ListNode* q = new ListNode(head->val - 1);
        q->next = head;

        ListNode* h = new ListNode();
        ListNode* p = h;

        ListNode* fast = head;
        ListNode* slow = q;
        int pre = head->val - 1;
        while (fast) {
            if (slow->val != pre && fast->val != slow->val) {
                p->next = slow;
                p = p->next;
            }
            pre = slow->val;
            slow = slow->next;
            fast = fast->next;
        }
        p->next = nullptr;
        if (slow->val != pre) p->next = slow;
        return h->next;
    }
};
```

[143. 重排链表](https://leetcode.cn/problems/reorder-list/description/)

```c++
class Solution {
public:
    ListNode* middle(ListNode* head) {
        if (!head) return head;
        ListNode* slow = head;
        ListNode* fast = head;
        while (fast->next && fast->next->next) {
            slow = slow->next;
            fast = fast->next->next;
        }
        return slow;
    }

    ListNode* reverse(ListNode* head) {
        if (!head) return head;
        if (!head->next) return head;
        auto end = reverse(head->next);
        head->next->next = head;
        head->next = nullptr;
        return end;
    }

    ListNode* merge(ListNode* l, ListNode* r) {
        ListNode* h = new ListNode();
        ListNode* p = h;
        while (l && r) {
            p->next = l;
            l = l->next;
            p = p->next;

            p->next = r;
            r = r->next;
            p = p->next;
        }
        p->next = l ? l : r;
        return h->next;
    }

    void reorderList(ListNode* head) {
        if (!head) return;
        ListNode* mid = middle(head);
        ListNode* l = head;
        ListNode* r = reverse(mid->next);
        mid->next = nullptr;
        head = merge(l, r);
    }
};
```

[148. 排序链表](https://leetcode.cn/problems/sort-list/description/)

```c++
class Solution {
public:
    ListNode* middle(ListNode* l, ListNode* r) {
        ListNode* slow = l;
        ListNode* fast = l;
        while (fast != r) {
            slow = slow->next;
            fast = fast->next;
            if (fast != r) fast = fast->next;
        }
        return slow;
    }

    ListNode* merge(ListNode* lhs, ListNode* rhs) {
        ListNode* h = new ListNode();
        ListNode* p = h;
        ListNode* l = lhs;
        ListNode* r = rhs;
        while (l && r) {
            if (l->val < r->val) {
                p->next = l;
                l = l->next;
            } else {
                p->next = r;
                r = r->next;
            }
            p = p->next;
        }
        p->next = l ? l : r;
        return h->next;
    }

    ListNode* f(ListNode* l, ListNode* r) {
        if (!l) return l;
        if (l->next == r) {
            l->next = nullptr;
            return l;
        }
        ListNode* mid = middle(l, r);
        auto lhs = f(l, mid);
        auto rhs = f(mid, r);
        return merge(lhs, rhs);
    }
    
    ListNode* sortList(ListNode* head) {
        if (!head) return head;
        return f(head, nullptr);
    }
};
```

[146. LRU 缓存](https://leetcode.cn/problems/lru-cache/description/)

```c++
class LRUCache {
public:
    class Node {
    public:
        int k, v;
        Node* prev;
        Node* next;
        Node(int k, int v) {
            this->k = k;
            this->v = v;
            this->prev = nullptr;
            this->next = nullptr;
        }
    };
    
    Node* head;
    map<int, Node*> mp;
    int cap; 
    int cnt;

    LRUCache(int capacity) {
        this->cap = capacity;
        this->cnt = 0;
        this->head = new Node(-1, -1);
        head->prev = head;
        head->next = head;
    }

    void insert(Node* head, Node* node) {
        node->next = head->next;
        head->next->prev = node;
        head->next = node;
        node->prev = head;
    }

    void remove(Node* node) {
        node->prev->next = node->next;
        node->next->prev = node->prev;
    }

    int get(int key) {
        if (!mp.count(key)) return -1;
        Node* node = mp[key];
        remove(node);
        insert(head, node);
        return node->v;
    }
    
    void put(int key, int value) {
        if (mp.count(key)) {
            Node* node = mp[key];
            node->v = value;
            remove(node);
            insert(head, node);
        } else {
            Node* node = new Node(key, value);
            if (cap == cnt) {
                Node* rear = head->prev;
                remove(rear);
                mp.erase(rear->k);
                cnt--;
            }
            insert(head, node);
            cnt++;
            mp[key] = node;
        }
    }
};
```

如果需要被构造的并非 LRU 而是 LFU，其思路将迥然不同。
