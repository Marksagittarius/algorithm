# 0x21 回溯

## 引言

回溯算法是最为基础的搜索算法，如何高效建模搜索空间与进行剪枝优化是使用回溯算法常面临的挑战。

## 例题

[78. 子集](https://leetcode.cn/problems/subsets/description/)

```c++
class Solution {
public:
    vector<vector<int> > ans;
    void f(vector<int>& nums, int i, int n, vector<int>& arr) {
        ans.push_back(arr);
        if (i == n) return;
        for (int j = i; j < n; j++) {
            arr.push_back(nums[j]);
            f(nums, j + 1, n, arr);
            arr.pop_back();
        }
    }
    vector<vector<int>> subsets(vector<int>& nums) {
        int n = nums.size();
        vector<int> arr;
        f(nums, 0, n, arr);
        return ans;
    }
};
```

[77. 组合](https://leetcode.cn/problems/combinations/description/)

```c++
class Solution {
public:
    vector<vector<int>> ans;
    void f(vector<int>& nums, int i, int k, int n, vector<int>& arr) {
        if (arr.size() == k) {
            ans.push_back(arr);
            return;
        }

        for (int j = i; j < n; j++) {
            arr.push_back(nums[j]);
            f(nums, j + 1, k, n, arr);
            arr.pop_back();
        }
    }
    vector<vector<int>> combine(int n, int k) {
        vector<int> nums(n);
        for (int i = 0; i < n; i++) nums[i] = i + 1;
        vector<int> arr;
        f(nums, 0, k, n, arr);
        return ans;
    }
};
```

[39. 组合总和](https://leetcode.cn/problems/combination-sum/description/)

```c++
class Solution {
public:
    vector<vector<int>> ans;
    void f(vector<int>& nums, int i, int n, int num, vector<int>& arr)  {
        if (num == 0) {
            ans.push_back(arr);
            return;
        }
        for (int j = i; j < n; j++) {
            for (int k = 1; num - k * nums[j] >= 0; k++) {
                for (int l = 0; l < k; l++) arr.push_back(nums[j]);
                f(nums, j + 1, n, num - k * nums[j], arr);
                for (int l = 0; l < k; l++) arr.pop_back();
            }
        }
    }
    vector<vector<int>> combinationSum(vector<int>& candidates, int target) {
        int n = candidates.size();
        sort(candidates.begin(), candidates.end());
        vector<int> arr;
        f(candidates, 0, n, target, arr);
        return ans;
    }
};
```

[40. 组合总和 II](https://leetcode.cn/problems/combination-sum-ii/description/https://leetcode.cn/problems/combination-sum-ii/description/)

```c++
class Solution {
public:
    vector<vector<int>> ans;
    vector<int> vis;
    void f(vector<int>& nums, int i, int n, int num, vector<int>& arr) {
        if (num == 0) {
            ans.push_back(arr);
            return;
        }
        for (int j = i; j < n; j++) {
            if (nums[j] > num) break;
            if (j >= 1 && nums[j] == nums[j - 1]) if (!vis[j - 1]) continue;
            vis[j] = 1;
            arr.push_back(nums[j]);
            f(nums, j + 1, n, num - nums[j], arr);
            vis[j] = 0;
            arr.pop_back();
        }
    }

    vector<vector<int>> combinationSum2(vector<int>& candidates, int target) {
        int n = candidates.size();
        vector<int> arr;
        vis = vector<int>(n);
        sort(candidates.begin(), candidates.end());
        f(candidates, 0, n, target, arr);
        return ans;
    }
};
```

class Solution {

            return;

        for (int i = 0; i < n; i++) {
            if (!vis[i]) {
                vis[i] = 1;
                arr.push_back(nums[i]);
                f(nums, n, arr);
                arr.pop_back();
                vis[i] = 0;
            }
        }
    }

        int n = nums.size();
        vis = vector<int>(n);
        f(nums, n, arr);

    }
};

[22. 括号生成](https://leetcode.cn/problems/IDBivT/description/?envType=study-plan-v2&envId=coding-interviews-special)

```c++
class Solution {
public:
    vector<string> ans;
    void f(int l, int r, int n, string str) {
        if (l == n && r == n) {
            ans.push_back(str);
            return;
        }
        if (l < n) f(l + 1, r, n, str + "(");
        if (l > r) f(l, r + 1, n, str + ")");
    }
    vector<string> generateParenthesis(int n) {
        f(0, 0, n, "");
        return ans;
    }
};
```

[52. N 皇后 II](https://leetcode.cn/problems/n-queens-ii/description/)

```c++
class Solution {
public:
    int ans = 0;
    vector<int> col;
    vector<int> lhs;
    vector<int> rhs;

    int totalNQueens(int n) {
        col = vector<int> (n, 0);
        lhs = vector<int> (2 * n - 1, 0);
        rhs = vector<int> (2 * n - 1, 0);
        f(0, n);
        return ans; 
    }

    void f(int x, int n) {
        if (x == n) {
            ans++;
            return;
        }

        for (int y = 0; y < n; y++) {
            if (col[y] || lhs[x - y + n - 1] || rhs[x + y]) 
                continue;

            col[y] = 1;
            lhs[x - y + n - 1] = 1;
            rhs[x + y] = 1;
            f(x + 1, n);
            col[y] = 0;
            lhs[x - y + n - 1] = 0;
            rhs[x + y] = 0;
        }
    }
};
```

[93. 复原 IP 地址](https://leetcode.cn/problems/restore-ip-addresses/description/)

```c++
class Solution {
public:
    vector<string> ans;
    void f(string& s, int i, vector<int>& arr) {
        int n = s.size();
        if (arr.size() > 4) return;
        if (i == n && arr.size() == 4) {
            string s = "";
            for (int i = 0; i < 4; i++) {
                s += to_string(arr[i]);
                if (i < 3) s += '.';
            }
            ans.push_back(s);
            return;
        }
        int num = 0;
        for (int j = i; j < n; j++) {
            num = num * 10 + (s[j] - '0');
            if (num > 255) break;
            arr.push_back(num);
            f(s, j + 1, arr);
            arr.pop_back();
            if (num == 0) break;
        }
    }
    vector<string> restoreIpAddresses(string s) {
        vector<int> arr;
        f(s, 0, arr);
        return ans;
    }
};
```
