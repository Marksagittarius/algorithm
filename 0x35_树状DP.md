# 0x35 树状 DP

## 引言

树状 DP 将数组或矩阵中的动态规划思想迁移到树上。

## 例题

[337. 打家劫舍 III](https://leetcode.cn/problems/house-robber-iii/description/)

```c++
class Solution {
public:
    int rob(TreeNode* root) {
       vector<int> ans = f(root);
       return max(ans[0], ans[1]);
    }

    vector<int> f(TreeNode* root) {
        if (!root) return {0, 0};
        vector<int> l = f(root->left);
        vector<int> r = f(root->right);
        int lhs = root->val + l[0] + r[0];
        int rhs = max(l[0], l[1]) + max(r[0], r[1]);
        return {rhs, lhs};
    }
};
```

[968. 监控二叉树](https://leetcode.cn/problems/binary-tree-cameras/)

```c++
class Solution {
public:
    struct T {
        int a, b, c;
    };

    T f(TreeNode* root) {
        if (!root) return { INT_MAX / 2, 0, 0 };
        auto [la, lb, lc] = f(root->left);
        auto [ra, rb, rc] = f(root->right);
        int a = lc + rc + 1;
        int b = min(a, min(la + rb, lb + ra));
        int c = min(a, lb + rb);
        return {a, b, c};
    }

    int minCameraCover(TreeNode* root) {
        auto [a, b, c] = f(root);
        return b;
    }
};
```
