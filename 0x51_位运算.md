# 0x51 位运算

## 引言

位运算是计算机科学的基石。

## 技巧

检查第 i 位是否为 1：(n >> i) & 1

将第 i 位设为 1：n | (1 << i)

将第 i 位设为 0：n & ~(1 << i)

将第 i 位取反：n ^ (1 << i)

提取最右边的 1（lowbit）：n & -n

消除最右边的 1：n & (n - 1)

```c++
int mulPowerOfTwo(int n, int m) { return n << m; }
int divPowerOfTwo(int n, int m) { return n >> m; }
int Abs(int n) { return (n ^ (n >> 31)) - (n >> 31); }
void swap(int& a, int& b) { a ^= b ^= a ^= b; }

int getBit(int a, int b) { return (a >> b) & 1; }
int setBit(int a, int b) { return a | (1 << b); }
int unsetBit(int a, int b) { return a & ~(1 << b); }
int flapBit(int a, int b) { return a ^ (1 << b); }
```

汉明权重

```c++
int f(int x) {
    int cnt = 0;
    while (x) {
        cnt++;
        x -= x & -x;
    }
    return cnt;
}
```

## 例题

[89. 格雷编码](https://leetcode.cn/problems/gray-code/)

```c++
class Solution {
public:
    vector<int> grayCode(int n) {
        vector<int> ans(1 << n);
        for (int i = 0; i < ans.size(); i++)
            ans[i] = (i >> 1) ^ i;
        return ans;
    }
};
```

[190. 颠倒二进制位](https://leetcode.cn/problems/reverse-bits/description/)

```c++
class Solution {
public:
    uint32_t reverseBits(uint32_t n) {
        uint32_t ans = 0;
        for (int i = 0; i < 32; i++) {
            ans <<= 1;
            ans += n & 1;
            n >>= 1;
        }
        return ans;
    }
};
```

[201. 数字范围按位与](https://leetcode.cn/problems/bitwise-and-of-numbers-range/description/)

```c++
class Solution {
public:
    int rangeBitwiseAnd(int left, int right) {
        while (left < right) right = right & (right - 1);
        return right;
    }
};
```

[421. 数组中两个数的最大异或值](https://leetcode.cn/problems/maximum-xor-of-two-numbers-in-an-array/)

```c++
class Solution {
class Trie {
public:
    vector<Trie*> children;
    Trie() {
        children = vector<Trie*>(2, nullptr);
    }
    
    void insert(int num) {
        Trie* trie = this;
        for (int i = 30; i >= 0; i--) {
            int bit = (num >> i) & 1;
            if (!trie->children[bit])
                trie->children[bit] = new Trie();
            trie = trie->children[bit];
        }
    }

    int search(int num) {
        Trie* trie = this;
        int ans = 0;
        for (int i = 30; i >= 0; i--) {
            int bit = (num >> i) & 1;
            bit = !trie->children[bit ^ 1] ? bit : bit ^ 1;
            ans += bit << i;
            trie = trie->children[bit];
        }
        return ans;
    }
};
public:
    int findMaximumXOR(vector<int>& nums) {
        int ans = 0;
        Trie* trie = new Trie();
        for (auto& num : nums) {
            trie->insert(num);
            ans = max(ans, num ^ trie->search(num));
        }

        return ans;
    }
};
```

[477. 汉明距离总和](https://leetcode.cn/problems/total-hamming-distance/description/)

```c++
class Solution {
public:
    int trie[32][2];
    int totalHammingDistance(vector<int>& nums) {
        int n = nums.size();
        for (auto& num : nums) {
            if (num == 0) {
                trie[0][0]++;
            } else {
                int idx = 0;    
                while (num) {
                    int m = num & 1;
                    trie[idx++][m]++;
                    num = (num >> 1);
                }
            }
        }
        int ans = 0;
        for (int i = 0; i < 32; i++) {
            int lhs = trie[i][0], rhs = trie[i][1];
            if (lhs == 0 && rhs == 0) break;
            if (rhs > 0) ans += rhs * (n - rhs);
        }
        return ans;
    }
};
```
