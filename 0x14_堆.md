# 0x14 堆

## 引言

堆是一种能够在指数时间复杂度内动态维护其根部单调性的数据结构。

在实践中常用优先队列结构以实现堆。

## 例题

[23. 合并 K 个升序链表](https://leetcode.cn/problems/merge-k-sorted-lists/description/)

```c++
class Solution {
public:
    struct cmp {
        bool operator()(const ListNode* a, const ListNode* b) const {
            return a->val > b->val;
        }
    };
    ListNode* mergeKLists(vector<ListNode*>& lists) {
        ListNode* head = new ListNode();
        ListNode* p = head;
        priority_queue<ListNode*, vector<ListNode*>, cmp> que;
        for (auto node : lists) if (node) que.push(node);
        while (!que.empty()) {
            auto top = que.top();
            que.pop();
            p->next = top;
            p = p->next;
            if (top->next) que.push(top->next);
        }
        return head->next;
    }
};
```

[295. 数据流的中位数](https://leetcode.cn/problems/find-median-from-data-stream/description/)

```c++
class MedianFinder {
public:
    priority_queue<int> lhs;
    priority_queue<int, vector<int>, greater<int> > rhs;
    MedianFinder() {}
    
    void addNum(int num) {
        if (lhs.size() == rhs.size()) {
            if (lhs.empty()) {
                rhs.push(num);
            } else {
                auto top = lhs.top();
                if (num > top) {
                    rhs.push(num);
                } else {
                    rhs.push(top);
                    lhs.pop();
                    lhs.push(num);
                }
            }
        } else {
            auto top = rhs.top();
            if (num > top) {
                rhs.pop();
                lhs.push(top);
                rhs.push(num);
            } else {
                lhs.push(num);
            }
        }
    }
    
    double findMedian() {
        if (lhs.size() == rhs.size())
            return 0.5 * (rhs.top() + lhs.top());
        return 1.0 * rhs.top();
    }
};
```

[218. 天际线问题](https://leetcode.cn/problems/the-skyline-problem/description/)

```c++
class Solution {
public:
    vector<vector<int>> getSkyline(vector<vector<int>>& buildings) {
        vector<vector<int>> ans;
        vector<pair<int, int>> arr;

        for (const auto& b : buildings) {
            arr.push_back({b[0], -b[2]});
            arr.push_back({b[1], b[2]});
        }
        sort(arr.begin(), arr.end());

        priority_queue<int> que;
        priority_queue<int> del;
        
        int prev = 0;
        que.push(prev);

        for (auto& it : arr) {
            int p = it.first;
            int h = it.second;

            if (h < 0) {
                que.push(-h);
            } else {
                del.push(h);
            }

            while (!del.empty() && del.top() == que.top()) {
                que.pop();
                del.pop();
            }

            int cur = que.top();
            if (cur != prev) {
                ans.push_back({p, cur});
                prev = cur;
            }
        }

        return ans;
    }
};
```

该题目的解法中使用了延迟删除技巧以实现优先队列中任意元素的删除。
