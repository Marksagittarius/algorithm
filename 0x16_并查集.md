# 0x16 并查集

## 引言

并查集是一种小而美的数据结构，能够有效地解决各类等价关系（即满足自反性，对称性与可传递性的关系）的抽象。

并查集的一般构造方式如下：

```c++
vector<int> q(n);
for (int i = 0; i < n; i++) q[i] = i;

int find(int x) {
    return q[x] == x ? q[x] : q[x] = find(q[x]);
}

void conn(int x, int y) {
    q[find(x)] = find(y);
}

bool check(int x, int y) {
    return find(x) == find(y);
}
```

## 例题

[547. 省份数量](https://leetcode.cn/problems/number-of-provinces/description/)

```c++
class Solution {
public:
    vector<int> f;
    int find(int x) {
        return f[x] == x ? x : f[x] = find(f[x]);
    }
    void conn(int x, int y) {
        f[find(x)] = find(y);
    }
    int findCircleNum(vector<vector<int>>& arr) {
        int n = arr.size();
        f = vector<int>(n);
        for (int i = 0; i < n; i++) f[i] = i;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (arr[i][j] && i != j) {
                    conn(i, j);
                }
            }
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (f[i] == i) {
                ans++;
            }
        }
        return ans;
    }
};
```

[684. 冗余连接](https://leetcode.cn/problems/redundant-connection/description/)

```c++
class Solution {
public:
    vector<int> f;
    int find(int x) {
        return f[x] == x ? x : f[x] = find(f[x]);
    }
    void conn(int x, int y) {
        f[find(x)] = find(y);
    }
    bool get(int x, int y) {
        return find(x) == find(y);
    }
    vector<int> findRedundantConnection(vector<vector<int>>& edges) {
        int n = edges.size();
        f = vector<int>(n + 1);
        for (int i = 0; i <= n; i++) f[i] = i;
        vector<vector<int>> ans;
        for (auto& e : edges) {
            int l = e[0], r = e[1];
            if (get(l, r)) return e;
            conn(l, r);
        }
        return edges[n - 1];
    }
};
```

[839. 相似字符串组](https://leetcode.cn/problems/similar-string-groups/description/)

```c++
class Solution {
public:
    vector<int> f;
    int find(int x) {
        return f[x] == x ? x : f[x] = find(f[x]);
    }
    void conn(int x, int y) {
        f[find(x)] = find(y);
    }
    int numSimilarGroups(vector<string>& strs) {
        int n = strs.size();
        int len = strs[0].size();
        f = vector<int>(n);
        for (int i = 0; i < n; i++) f[i] = i;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < i; j++) {
                auto lhs = strs[i];
                auto rhs = strs[j];
                int cnt = 0;
                for (int k = 0; k < len; k++) {
                    if (lhs[k] != rhs[k]) {
                        cnt++;
                    }
                }
                if (cnt == 2 || cnt == 0) conn(i, j);
            }
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (f[i] == i) {
                ans++;
            }
        }
        return ans;
    }
};
```

[352. 将数据流变为多个不相交区间](https://leetcode.cn/problems/data-stream-as-disjoint-intervals/description/)

```c++

class SummaryRanges {
private:
    unordered_map<int, int> f;
    map<int, int> mp;
    set<int> st;
    int find(int x) {
        if (!f.count(x)) f[x] = x;
        return x == f[x] ? x : f[x] = find(f[x]);
    }

    void conn(int x, int y) {
        int xx = find(x);
        int yy = find(y);
        if (xx < yy) {
            f[yy] = xx;
            mp[xx] = mp[yy];
            st.erase(yy);
        } else {
            conn(yy, xx);
        }
    }

public:
    SummaryRanges() {
    }
    
    void addNum(int val) {
        if (f.count(val)) return;
        f[val] = val;
        mp[val] = val;
        if (f.count(val - 1)) {
            conn(val - 1, val);
        }
        if (f.count(val + 1)) {
            conn(val, val + 1);
        }
        if (val == find(val)) {
            st.insert(val);
        }
    }
    
    vector<vector<int>> getIntervals() {
        vector<vector<int>> ans;
        for (auto& val : st) {
            if (val == find(val)) {
                ans.push_back({val, mp[val]});
            }
        }
        return ans;
    }
};
```

[765. 情侣牵手](https://leetcode.cn/problems/couples-holding-hands/description/)

```c++
class Solution {
public:
    int cnt;
    vector<int> f;
    int find(int x) {
        return x == f[x] ? x : f[x] = find(f[x]);
    }

    void conn(int x, int y) {
        int xx = find(x);
        int yy = find(y);
        if (xx != yy) {
            f[xx] = yy;
            cnt--;
        }
    }

    int minSwapsCouples(vector<int>& row) {
        int n = row.size() / 2;
        cnt = n;
        f = vector<int>(n, 0);
        for (int i = 0; i < n; i++) f[i] = i;
        for (int i = 0; i < 2 * n; i += 2) {
            conn(row[i] / 2, row[i + 1] / 2);
        }
        return n - cnt;
    }
};
```

该题目引入了置换环的思想，其一般构造方式如下：

```c++
class Solution {
public:
    int cnt;
    vector<int> f;

    int find(int x) {
        return x == f[x] ? x : f[x] = find(f[x]);
    }

    void conn(int x, int y) {
        int xx = find(x);
        int yy = find(y);
        if (xx != yy) {
            f[xx] = yy;
            cnt--;
        }
    }

    int minSwapsGeneral(vector<int>& arr, vector<int>& target) {
        int n = arr.size();
        cnt = n;
        f = vector<int>(n);
        for (int i = 0; i < n; i++) f[i] = i;

        unordered_map<int, int> mp;
        for (int i = 0; i < n; i++) {
            mp[target[i]] = i;
        }

        for (int i = 0; i < n; i++) {
            conn(i, mp[arr[i]]);
        }

        return n - cnt;
    }
};
```
