# 0x13 队列

## 引言

栈是一种后进先出（LIFO）的数据结构，常被用于广度优先搜索等场景中。

[102. 二叉树的层序遍历](https://leetcode.cn/problems/binary-tree-level-order-traversal/description/)

```c++
class Solution {
public:
    vector<vector<int>> levelOrder(TreeNode* root) {
        vector<vector<int>> ans;
        queue<TreeNode*> que;
        if (root) que.push(root);
        while (!que.empty()) {
            int l = que.size();
            vector<int> arr;
            for (int i = 0; i < l; i++) {
                auto node = que.front();
                que.pop();
                arr.push_back(node->val);
                if (node->left) que.push(node->left);
                if (node->right) que.push(node->right);
            }
            ans.emplace_back(arr);
        }
        return ans;
    }
};
```

## 单调队列

单调队列是一种特殊的队列，能够高效地解决「滑动窗口最大值」问题，其一般构造方式如下：

```c++
deque<int> que;

void pop(int a) {
    if (!que.empty() && a == que.front())
        que.pop_front();
}

void push(int a) {
    while (!que.empty() && a > que.back())
        que.pop_back();
    que.push_back(a);
}
```

[239. 滑动窗口最大值](https://leetcode.cn/problems/sliding-window-maximum/description/)

```c++
class Solution {
public:
    deque<int> que;

    void push(int x) {
        while (!que.empty() && x > que.back()) que.pop_back();
        que.push_back(x);
    }
    
    void pop(int x) {
        if (!que.empty() && x == que.front()) que.pop_front();
    }

    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        vector<int> ans;
        int n = nums.size();
        for (int i = 0; i < k; i++) push(nums[i]);
        ans.push_back(que.front());
        for (int i = k; i < n; i++) {
            pop(nums[i - k]);
            push(nums[i]);
            ans.push_back(que.front());
        }
        return ans;
    }
};
```

单调队列亦可以与滑动窗口组合使用。

[2398. 预算内的最多机器人数目](https://leetcode.cn/problems/maximum-number-of-robots-within-budget/description/)

```c++
class Solution {
public:
    deque<int> que;

    void push(int x) {
        while (!que.empty() && x > que.back()) que.pop_back();
        que.push_back(x);
    }

    void pop(int x) {
        if (!que.empty() && x == que.front())
            que.pop_front();
    }

    int maximumRobots(vector<int>& chargeTimes, vector<int>& runningCosts, long long budget) {
        int n = chargeTimes.size();
        int ans = 0;
        int l = 0;
        long long sum = 0;
        for (int r = 0; r < n; r++) {
            sum += runningCosts[r];
            push(chargeTimes[r]);
            int k = r - l + 1;
            while (que.front() + k * sum > budget) {
                sum -= runningCosts[l];
                pop(chargeTimes[l]);
                k--;
                l++;
            }
            ans = max(ans, k);
        }
        return ans;
    }
};
```

单调队列亦可同单调栈配合。

[862. 和至少为 K 的最短子数组](https://leetcode.cn/problems/shortest-subarray-with-sum-at-least-k/description/)

```c++
class Solution {
public:
    int shortestSubarray(vector<int>& nums, int k) {
        int n = nums.size();
        int ans = n + 1;
        deque<int> que;
        vector<long> pre(n + 1);
        for (int i = 1; i <= n; i++) 
            pre[i] = pre[i - 1] + nums[i - 1];
        for (int i = 0; i <= n; i++) {
            long sum = pre[i];
            while (!que.empty() && sum - pre[que.front()] >= k) {
                ans = min(ans, i - que.front());
                que.pop_front();
            }
            while (!que.empty() && pre[que.back()] >= sum) {
                que.pop_back();
            }
            que.push_back(i);
        }
        return ans <= n ? ans : -1;
    }
};
```
