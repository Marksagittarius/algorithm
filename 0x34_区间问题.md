# 0x34 区间问题

## 引言

区间动态规划是以区间两端点为范围进行的动态规划。

## 例题

[312. 戳气球](https://leetcode.cn/problems/burst-balloons/description/)

```c++
class Solution {
public:
    int maxCoins(vector<int>& nums) {
        int n = nums.size();
        vector<int> g(n + 2);
        g[0] = g[n + 1] = 1;
        for (int i = 0; i < n; i++) g[i + 1] = nums[i];
        vector<vector<int> > f(n + 2, vector<int>(n + 2));
        for (int i = n; i >= 0; i--) {
            for (int j = i + 1; j <= n + 1; j++) {
                for (int k = i + 1; k < j; k++) {
                    f[i][j] = max(f[i][j], f[i][k] + f[k][j] + g[i] * g[j] * g[k]);
                }
            }
        }
        return f[0][n + 1];
    }
};
```

[1547. 切棍子的最小成本](https://leetcode.cn/problems/minimum-cost-to-cut-a-stick/description/)

```c++
class Solution {
public:
    int minCost(int n, vector<int>& cuts) {
        int m = cuts.size();
        sort(cuts.begin(), cuts.end());
        cuts.insert(cuts.begin(), 0);
        cuts.push_back(n);
        vector<vector<int>> f(m + 2, vector<int>(m + 2));
        for (int i = 0; i <= m + 1; i++) f[i][i] = 0;
        for (int i = m; i >= 0; i--) {
            for (int j = i + 2; j <= m + 1; j++) {
                f[i][j] = INT_MAX;
                for (int k = i + 1; k < j; k++) {
                    f[i][j] = min(f[i][j], f[i][k] + f[k][j]);
                }
                f[i][j] += cuts[j] - cuts[i];
            }
        }
        return f[0][m + 1];
    }
};
```
