# 0x24 拓扑排序

## 引言

拓扑排序并非一种排序算法。

## 例题

[207. 课程表](https://leetcode.cn/problems/course-schedule/)

```c++
class Solution {
public:
    bool canFinish(int numCourses, vector<vector<int>>& prerequisites) {
        int n = numCourses;
        vector<int> deg(n, 0);
        unordered_map<int, vector<int> >g;
        queue<int> que;
        
        for (auto pre : prerequisites) {
            g[pre[1]].push_back(pre[0]);
            deg[pre[0]]++;
        }

        for (int i = 0; i < n; i++)
            if (deg[i] == 0)
                que.push(i);

        int cnt = 0; 
        while (!que.empty()) {
            auto it = que.front();
            que.pop();
            cnt++;
            for (auto n : g[it])
                if (--deg[n] == 0)
                    que.push(n);
        }
        return cnt == n;
    }
};
```

[329. 矩阵中的最长递增路径](https://leetcode.cn/problems/longest-increasing-path-in-a-matrix/description/)

```c++
class Solution {
public:
    vector<vector<int> > deg;
    int dx[4] = {0, 1, 0, -1};
    int dy[4] = {1, 0, -1, 0};
    int longestIncreasingPath(vector<vector<int>>& matrix) {
        int m = matrix.size();
        int n = matrix[0].size();
        deg = vector<vector<int> >(m, vector<int>(n));
        queue<pair<int, int> > que;
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < 4; k++) {
                    int xx = i + dx[k];
                    int yy = j + dy[k];
                    if (xx >= 0 && yy >= 0 && xx < m && yy < n) {
                        if (matrix[xx][yy] > matrix[i][j]) {
                            deg[i][j]++;
                        }
                    }
                }
                if (deg[i][j] == 0) que.push({i, j});
            }
        }
        int ans = 0;
        while (!que.empty()) {
            ans++;
            int l = que.size();
            for (int j = 0; j < l; j++) {
                auto it = que.front();
                que.pop();
                int x = it.first;
                int y = it.second;
                for (int i = 0; i < 4; i++) {
                    int xx = x + dx[i];
                    int yy = y + dy[i];
                    if (xx >= 0 && yy >= 0 && xx < m && yy < n) {
                        if (matrix[xx][yy] < matrix[x][y]) {
                            if (--deg[xx][yy] == 0)
                                que.push({xx, yy});
                        }
                    }
                }
            }
        }
        return ans;
    }
};
```

[LCR 115. 序列重建](https://leetcode.cn/problems/ur2n8P/description/?envType=study-plan-v2&envId=coding-interviews-special)

```c++
class Solution {
public:
    bool sequenceReconstruction(vector<int>& nums, vector<vector<int>>& sequences) {
        int n = nums.size();
        int cnt = 0;
        vector<vector<int> > g(n + 1, vector<int>(0));
        vector<int> vis(n + 1);
        vector<int> deg(n + 1);
        set<pair<int, int> > st;
        queue<int> que;
        for (auto& seq : sequences) {
            int l = seq.size();
            if (l == 1) {
                vis[seq[0]] = 1;
                continue;
            }
            for (int i = 1; i < l; i++) {
                vis[seq[i]] = vis[seq[i - 1]] = 1;
                if (st.count({seq[i], seq[i - 1]}))
                    continue;
                g[seq[i]].push_back(seq[i - 1]);
                deg[seq[i - 1]]++;
            }
        }
        for (int i = 1; i <= n; i++) {
            if (!vis[i]) return false;
            if (deg[i] == 0) {
                que.push(i);
            }
        }
        vis = vector<int>(n + 1);
        while (!que.empty()) {
            if (que.size() > 1) return false;
            int i = que.front();
            que.pop();
            if (vis[i]) continue;
            vis[i] = 1;
            cnt++;
            for (auto& j : g[i]) {
                if (--deg[j] == 0) {
                    que.push(j);
                }
            }
        }
        return cnt == n;
    }
};
```
