# 0x07 排序

## 引言

排序是优化算法问题求解过程的常用手段，许多经典的排序问题背后的思想也能够指导题目的求解。

## 快速排序

快速排序是一种快速的排序。

```c++
void quick_sort(int q[], int l, int r) {
    if (l >= r)
        return;

    int x = q[l];
    int i = l - 1;
    int j = r + 1;
    
    while (i < j) {
        do {
            i++;
        } while (q[i] < x);

        do {
            j++;
        } while (q[j] > x);
        
        if (i < j) swap(q[i], q[j]);
    }

    quick_sort(q, l, j);
    quick_sort(q, j + 1, r);
}
```

在 c++ 编程语言中常通过 `sort()` 函数以实现高速运转的快速排序。

[406. 根据身高重建队列](https://leetcode.cn/problems/queue-reconstruction-by-height/description/)

```c++
class Solution {
public:
    vector<vector<int>> reconstructQueue(vector<vector<int>>& people) {
        sort(people.begin(), people.end(), [&](auto& l, auto& r){
            return l[0] > r[0] || (l[0] == r[0]) && l[1] < r[1];
        });
        vector<vector<int>> ans;
        for (auto& p : people)
            ans.insert(ans.begin() + p[1], p);
        return ans;
    }
};
```

## 归并排序

归并排序是一种不仅快速而且稳定的排序。

```c++
void merge_sort(int q[], int l, int r) {
    if (l >= r)
        return;

    int mid = (l + r) / 2;
    merge_sort(q, l, mid);
    merge_sort(q, mid + 1, r);

    int k = 0;
    int i = l;
    int j = mid + 1;

    while (i <= mid && j <= r)
        if (q[i] <= q[j])
            temp[k++] = q[i++];
        else
            temp[k++] = q[j++];
    
    while (i <= mid) temp[k++] = q[i++];
    while (j <= r) temp[k++] = q[j++];

    for (int i = l, j = 0; i <= r; i++, j++)
        q[i] = temp[j];
}
```

归并排序体现了「分而治之」的伟大思想。

归并排序常被用来求解逆序对问题及其变体。

[315. 计算右侧小于当前元素的个数](https://leetcode.cn/problems/count-of-smaller-numbers-after-self/description/)

```c++
class Solution {
public:
    struct Num {
        int val;
        int idx;
    };
    vector<int> ans;
    vector<Num> tmp;
    vector<int> countSmaller(vector<int>& nums) {
        int n = nums.size();
        ans = vector<int>(n);
        tmp = vector<Num>(n);
        vector<Num> arr;
        for (int i = 0; i < n; i++) arr.push_back({nums[i], i});
        mergeSort(arr, 0, n - 1);
        return ans;
    }

    void mergeSort(vector<Num>& nums, int l, int r) {
        if (l >= r) return;
        int mid = (l + r) / 2;
        mergeSort(nums, l, mid);
        mergeSort(nums, mid + 1, r);

        int i = l, j = mid + 1, k = l;
        while (i <= mid && j <= r) {
            int lhs = nums[i].val;
            int rhs = nums[j].val;
            if (lhs <= rhs) {
                ans[nums[i].idx] += j - mid - 1;
                tmp[k++] = nums[i++];
            } else {
                tmp[k++] = nums[j++];
            }
        }
        while (i <= mid) {
            ans[nums[i].idx] += j - mid - 1;
            tmp[k++] = nums[i++];
        }
        while (j <= r) tmp[k++] = nums[j++];
        for (k = l; k <= r; k++) nums[k] = tmp[k];
    }
};
```

[493. 翻转对](https://leetcode.cn/problems/reverse-pairs/description/)

```c++
class Solution {
public:
    int ans = 0;
    vector<long> tmp;
    vector<long> arr;
    int reversePairs(vector<int>& nums) {
        int n = nums.size();
        tmp = vector<long>(n);
        for (auto& num : nums) arr.push_back((long) num * 2);
        mergeSort(nums, 0, n - 1);
        return ans;
    }

    void mergeSort(vector<int>& nums, int l, int r) {
        if (l >= r) return;
        int mid = (l + r) / 2;
        mergeSort(nums, l, mid);
        mergeSort(nums, mid + 1, r);
        int i = l, j = mid + 1, k = l;
        while (i <= mid && j <= r) {
            if (nums[i] <= arr[j]) {
                ans += j - mid - 1;
                i++; 
            } else {
                j++;
            }
        }
        while (i <= mid) {
            ans += j - mid - 1;
            i++;
        }
        i = l, j = mid + 1;
        while (i <= mid && j <= r) {
            if (nums[i] <= nums[j]) {
                tmp[k++] = nums[i++];
            } else {
                tmp[k++] = nums[j++];
            }
        }
        while (i <= mid) tmp[k++] = nums[i++];
        while (j <= r) tmp[k++] = nums[j++];
        for (k = l; k <= r; k++) {
            nums[k] = tmp[k];
            arr[k] = tmp[k] * 2;
        }
    }
};
```

[327. 区间和的个数](https://leetcode.cn/problems/count-of-range-sum/description/)

```c++
class Solution {
public:
    int ans = 0;
    vector<long long> tmp;
    int countRangeSum(vector<int>& nums, int lower, int upper) {
        int n = nums.size();
        vector<long long> arr(n + 1);
        tmp = vector<long long>(n + 1);
        for (int i = 1; i <= n; i++) {
            arr[i] = arr[i - 1] + nums[i - 1];
        }
        mergeSort(arr, 0, n, lower, upper);
        return ans;
    }

    void mergeSort(vector<long long>& nums, int l, int r, int ll, int rr) {
        if (l >= r) return;
        int mid = (l + r) / 2;
        mergeSort(nums, l, mid, ll, rr);
        mergeSort(nums, mid + 1, r, ll, rr);

        int lhs = mid + 1;
        int rhs = mid + 1;
        for (int i = l; i <= mid; i++) {
            while (lhs <= r && nums[lhs] - nums[i] < ll) lhs++;
            while (rhs <= r && nums[rhs] - nums[i] <= rr) rhs++;
            ans += rhs - lhs;
        }

        int i = l, j = mid + 1, k = l;

        while (i <= mid && j <= r) {
            if (nums[i] <= nums[j]) {
                tmp[k++] = nums[i++];
            } else {
                tmp[k++] = nums[j++];
            }
        }
        while (i <= mid) tmp[k++] = nums[i++];
        while (j <= r) tmp[k++] = nums[j++];
        for (k = l; k <= r; k++) {
            nums[k] = tmp[k];
        }
    }
};
```
