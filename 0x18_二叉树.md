# 0x18 二叉树

## 引言

二叉树是一种较链表更为复杂的链式存储结构。

## 例题

[104. 二叉树的最大深度](https://leetcode.cn/problems/maximum-depth-of-binary-tree/submissions/678718127/)

```c++
class Solution {
public:
    int ans = 0;
    int maxDepth(TreeNode* root) {
        f(root, 1);
        return ans;
    }

    void f(TreeNode* root, int num) {
        if (!root) return;
        ans = max(ans, num);
        f(root->left, num + 1);
        f(root->right, num + 1);
    }
};
```

[105. 从前序与中序遍历序列构造二叉树](https://leetcode.cn/problems/construct-binary-tree-from-preorder-and-inorder-traversal/description/)

```c++
class Solution {
public:
    TreeNode* buildTree(vector<int>& preorder, vector<int>& inorder) {
        int n = preorder.size();
        return f(preorder, inorder, 0, n - 1, 0, n - 1);
    }

    TreeNode* f(vector<int>& pre, vector<int>& in, int l1, int r1, int l2, int r2) {
        if (l1 > r1 || l2 > r2) return nullptr;
        int n = pre[l1];
        int l = -1;
        TreeNode* root = new TreeNode(n);
        for (int i = l2; i <= r2; i++) {
            if (in[i] == n) {
                l = i;
                break;
            }
        }
        root->left = f(pre, in, l1 + 1, l1 + l - l2, l2, l - 1);
        root->right = f(pre, in, l1 + l - l2 + 1, r1, l + 1, r2);
        return root;
    }
};
```

[112. 路径总和](https://leetcode.cn/problems/path-sum/description/)

```c++
class Solution {
public:
    bool hasPathSum(TreeNode* root, int targetSum) {
        return f(root, targetSum);
    }

    bool f(TreeNode* root, int num) {
        if (!root) return false;
        int val = root->val;
        if (!root->left && !root->right) return val == num;
        return f(root->left, num - val) || f(root->right, num - val);
    }
};
```

[113. 路径总和 II](https://leetcode.cn/problems/path-sum-ii/description/https://leetcode.cn/problems/path-sum-ii/description/)

```c++
class Solution {
public:
    vector<vector<int>> ans;
    vector<vector<int>> pathSum(TreeNode* root, int targetSum) {
        vector<int> arr;
        f(root, targetSum, arr);
        return ans;
    }

    void f(TreeNode* root, int num, vector<int>& arr) {
        if (!root) return;
        int val = root->val;
        arr.push_back(val);
        if (!root->left && !root->right) {
            if (val == num) {
                ans.push_back(arr);
                arr.pop_back();
                return;
            }
        }
        f(root->left, num - val, arr);
        f(root->right, num - val, arr);
        arr.pop_back();
    }
};
```

[437. 路径总和 III](https://leetcode.cn/problems/path-sum-iii/description/)

class Solution {
public:
    int pathSum(TreeNode* root, int targetSum) {
        if (!root) return 0;
        return f(root, targetSum) + pathSum(root->left, targetSum) + pathSum(root->right, targetSum);
    }

    long long f(TreeNode* root, long long num) {
        if (!root) return 0;
        int val = root->val;
        return (val == num) + f(root->left, num - val) + f(root->right, num - val);
    }
};

[124. 二叉树中的最大路径和](https://leetcode.cn/problems/binary-tree-maximum-path-sum/description/)

```c++
class Solution {
public:
    int ans = INT_MIN;
    int maxPathSum(TreeNode* root) {
        f(root);
        return ans;
    }

    int f(TreeNode* root) {
        if (!root) return 0;
        int val = root->val;
        int lhs = max(0, f(root->left));
        int rhs = max(0, f(root->right));
        ans = max(ans, val + lhs + rhs);
        return max(lhs, rhs) + val;
    }
};
```

[114. 二叉树展开为链表](https://leetcode.cn/problems/flatten-binary-tree-to-linked-list/description/)

```c++
class Solution {
public:
    void flatten(TreeNode* root) {
        f(root);
    }

    pair<TreeNode*, TreeNode*> f(TreeNode* root) {
        if (!root) return {nullptr, nullptr};
        auto lhs = f(root->left);
        auto rhs = f(root->right);
        root->left = nullptr;
        root->right = lhs.first;
        if (lhs.second) {
            lhs.second->left = nullptr;
            lhs.second->right = rhs.first;
        } else {
            root->right = rhs.first;
        }

        return {root, rhs.second ? rhs.second : (lhs.second ? lhs.second : root)};
    }
};
```

[236. 二叉树的最近公共祖先](https://leetcode.cn/problems/lowest-common-ancestor-of-a-binary-tree/description/)

```c++
class Solution {
public:
    TreeNode* lowestCommonAncestor(TreeNode* root, TreeNode* p, TreeNode* q) {
        if (!root || p == root || q == root) return root;
        auto lhs = lowestCommonAncestor(root->left, p, q);
        auto rhs = lowestCommonAncestor(root->right, p, q);
        if (lhs && !rhs) return lhs;
        if (rhs && !lhs) return rhs;
        if (lhs && rhs) return root;
        return NULL;
    }
};
```

[297. 二叉树的序列化与反序列化](https://leetcode.cn/problems/serialize-and-deserialize-binary-tree/)

```c++
class Codec {
public:
    string serialize(TreeNode* root) {
        vector<string> arr;
        encode(root, arr);
        string ans = "";
        for (auto& s : arr) {
            ans += s;
            ans += "|";
        }
        return ans;
    }

    void encode(TreeNode* root, vector<string>& arr) {
        if (!root) {
            arr.push_back("#");
            return;
        }
        arr.push_back(to_string(root->val));
        encode(root->left, arr);
        encode(root->right, arr);
    }

    const int NONE = 114514;
    
    bool digit(char& ch) {
        return ch >= '0' && ch <= '9';
    }

    TreeNode* deserialize(string data) {
        queue<int> arr;      
        int num = 0;
        int flag = 1;
        for (auto& ch : data) {
            if (ch == '|') {
                arr.push(flag * num);
                num = 0;
                flag = 1;
            } else if (ch == '#') {
                num = NONE;
            } else if (digit(ch)) {
                num = num * 10 + (int)(ch - '0');
            } else if (ch == '-') {
                flag = -1;
            }
        }
        return decode(arr);
    }

    TreeNode* decode(queue<int>& arr) {
        if (arr.empty()) {
            return nullptr;
        }
        if (arr.front() == NONE) {
            arr.pop();
            return nullptr;
        }
        int num = arr.front();
        arr.pop();
        TreeNode* root = new TreeNode(num);
        root->left = decode(arr);
        root->right = decode(arr);
        return root;
    }
};
```
